    reflow.claims¶

    reflow.claims ¶

    Exclusive resources: at most one active run per target.

    A workflow declares which parameters name resources that two runs must never use at the same time, typically an output location::

    wf = Workflow("era5-healpix", exclusive=["target"])

    When a run starts, reflow claims the normalised values of those parameters in the manifest store. The check and the claim happen in one transaction, so of two runs started at the same moment only one gets a key. Keys are global: a run of another workflow that claims the same target conflicts too. A claim is released when its run finishes, fails or is cancelled.

    If a run dies without finishing (its dispatch job was cancelled or killed along with its tasks), its claim would block the target forever. So before a conflict is reported, the holding run is checked with Executor.job_status, which asks the live queue of any backend: if neither its dispatch jobs nor its task jobs are queued or running, the run is marked FAILED and its claims are released. If the scheduler cannot tell, the claim is respected and the conflict reported: resolve it with cancel.

    Claims only protect runs that use the same manifest store.

    def normalise_key(value: Any) -> str:
    """Return the canonical form of a resource value.

    URLs (``s3://bucket/x/``) lose trailing slashes; anything else is
    treated as a local path and resolved, so ``./out`` and the absolute
    path of the same directory are the same key.
    """
    text = os.fspath(value) if isinstance(value, os.PathLike) else str(value)
    text = text.strip()
    if _URL.match(text):
        return text.rstrip("/")
    return str(Path(text).expanduser().resolve(strict=False))

    def resource_keys(
    spec: ExclusiveSpec | None, parameters: Mapping[str, Any]
) -> dict[str, str]:
    """Map normalised keys to labels such as ``"target=/scratch/out"``.

    Raises
    ------
    ValueError
        If a named parameter is missing from *parameters*.

    """
    if spec is None:
        return {}
    keys: dict[str, str] = {}
    if callable(spec):
        for value in spec(parameters):
            key = normalise_key(value)
            keys[key] = key
        return keys
    for name in spec:
        if name not in parameters or parameters[name] is None:
            raise ValueError(
                f"exclusive parameter {name!r} has no value; it must be a "
                "parameter of the workflow"
            )
        value = parameters[name]
        values = value if isinstance(value, (list, tuple, set)) else [value]
        for item in values:
            key = normalise_key(item)
            keys[key] = f"{name}={key}"
    return keys

    def run_liveness(
    run_id: str, store: Store, executor: Executor, claim: Mapping[str, Any]
) -> Liveness:
    """Check whether *run_id* (holding *claim*) is still active.

    Only reads: neither the run nor its instances are changed, because
    the holder may belong to another user whose result files cannot be
    read from here.
    """
    run = store.get_run(run_id)
    if run is None or run.get("status") != RunState.RUNNING.value:
        return Liveness(False, "the run is no longer running")

    pid = int(claim.get("pid") or 0)
    if pid:
        host = str(claim.get("host") or "")
        if host != socket.gethostname():
            return Liveness(
                None, f"it runs locally on host {host} (pid {pid}); cannot check"
            )
        if _pid_alive(pid):
            return Liveness(True, f"it runs locally here (pid {pid})")
        return Liveness(False, f"its local process {pid} is gone")

    # A running run always has something in the scheduler: its newest
    # dispatch job waits (afterany) for the task jobs of the last pass,
    # and task jobs may outlive a cancelled dispatch job.  `job_status`
    # asks the live queue, so this works on every backend.
    dispatch = list(reversed(store.list_jobs(run_id, "dispatch")))
    tasks = list(
        dict.fromkeys(
            str(inst["job_id"])
            for inst in store.list_task_instances(
                run_id, states=[TaskState.SUBMITTED, TaskState.RUNNING]
            )
            if inst.get("job_id")
        )
    )
    if not dispatch and not tasks:
        return Liveness(None, "no scheduler jobs are recorded for it")
    unknown = False
    for what, job_ids in (
        ("a dispatch pass is queued or running", dispatch),
        ("task jobs are queued or running", tasks),
    ):
        if not job_ids:
            continue
        state = executor.jobs_status(job_ids)
        if state.active:
            return Liveness(True, what)
        unknown = unknown or state.status is JobStatus.UNKNOWN
    if unknown:
        return Liveness(
            None, "the scheduler cannot tell whether its jobs are still running"
        )
    return Liveness(False, "none of its jobs is queued or running any more")

    def claim(
    store: Store,
    run_id: str,
    keys: Mapping[str, str],
    executor: Executor,
    *,
    workflow: str,
    local: bool = False,
) -> None:
    """Claim *keys* for *run_id*, freeing claims of dead runs on the way.

    Raises
    ------
    RunConflictError
        If a key is held by a run that is (or may be) still active.

    """
    if not keys:
        return
    owner = owner_info(local)
    # Each round either succeeds, reports live holders, or frees at least
    # one dead holder; a few rounds cover holders freed concurrently.
    for _ in range(5):
        held = store.claim_resources(run_id, dict(keys), owner)
        if not held:
            return
        conflicts: list[Conflict] = []
        freed = False
        for key, row in held.items():
            holder = str(row["run_id"])
            liveness = run_liveness(holder, store, executor, row)
            if liveness.alive is False:
                logger.warning(
                    "Run %s held %s but is no longer active (%s); releasing it",
                    holder,
                    keys[key],
                    liveness.detail,
                )
                run = store.get_run(holder)
                if run is not None and run.get("status") == RunState.RUNNING.value:
                    store.update_run_status(holder, RunState.FAILED)
                store.release_resources(holder)
                freed = True
                continue
            conflicts.append(Conflict(key, keys[key], holder, liveness.detail))
        if conflicts:
            raise RunConflictError(
                _conflict_message(workflow, conflicts, store), conflicts
            )
        if not freed:
            break
    raise RunConflictError(
        f"Could not claim {', '.join(keys.values())} for a new run of "
        f"{workflow!r}: other runs keep claiming it.",
        [],
    )

    @contextmanager
def graceful_shutdown() -> Generator[None, None, None]:
    """Context manager that converts SIGTERM/SIGINT to TaskInterrupted.

    Restores the original signal handlers on exit.

    Examples
    --------
    >>> with graceful_shutdown():
    ...     result = expensive_computation()

    """
    old_sigterm = signal.getsignal(signal.SIGTERM)
    old_sigint = signal.getsignal(signal.SIGINT)

    signal.signal(signal.SIGTERM, _signal_handler)
    signal.signal(signal.SIGINT, _signal_handler)

    try:
        yield
    finally:
        signal.signal(signal.SIGTERM, old_sigterm)
        signal.signal(signal.SIGINT, old_sigint)

    max_submit_jobs property ¶

    Cap on array tasks submitted per wave.

    Set this to your scheduler's per-user job-submit limit (e.g. Slurm AssocMaxSubmitJobLimit). When set, a large array task is submitted in waves of at most this many indices; the remainder stay PENDING and are submitted by the dependency-triggered follow-up dispatch as each wave drains, so the number of queued jobs never exceeds the cap. None (the default) disables capping and submits the whole array at once. Leave a little headroom below the true limit for the follow-up dispatch job itself and any concurrent singleton tasks.

    This is a reflow-internal scheduling cap, not a scheduler flag, so it is a plain [executor] key rather than a submit_options entry (those are rendered onto the sbatch command line).

    Config: [executor] max_submit_jobs; env: REFLOW_MAX_SUBMIT_JOBS.

    Uses scheduler-agnostic canonical keys (partition, account, etc.) which the active executor normalises to its native vocabulary at submission time.

    def dump_value(self, value: Any) -> Any:
    """Convert a Python value into a JSON-safe representation."""
    if isinstance(value, Enum):
        return {
            self.marker_key: True,
            _TYPE: "enum",
            _CLASS: self._qualname(type(value)),
            _VALUE: self.dump_value(value.value),
        }

    if isinstance(value, self._json_scalar_types):
        return value

    if isinstance(value, Path):
        return self._tag("path", str(value))

    if isinstance(value, datetime):
        return self._tag("datetime", value.isoformat())

    if isinstance(value, UUID):
        return self._tag("uuid", str(value))

    if is_dataclass(value) and not isinstance(value, type):
        payload = {
            name: self.dump_value(getattr(value, name))
            for name in (field.name for field in fields(value))
        }
        return {
            self.marker_key: True,
            _TYPE: "dataclass",
            _CLASS: self._qualname(type(value)),
            _FIELDS: payload,
        }

    if isinstance(value, tuple):
        return self._tag("tuple", [self.dump_value(item) for item in value])

    if isinstance(value, list):
        return [self.dump_value(item) for item in value]

    if isinstance(value, dict):
        bad_key = next((key for key in value if not isinstance(key, str)), None)
        if bad_key is not None:
            raise TypeError(
                "Manifest dictionaries must use string keys; "
                f"got key {bad_key!r} of type {type(bad_key).__name__}."
            )
        return {key: self.dump_value(item) for key, item in value.items()}

    raise TypeError(
        "Unsupported manifest value type: "
        f"{type(value).__module__}.{type(value).__qualname__}"
    )

    def load_value(self, value: Any) -> Any:
    """Reconstruct a Python value from a JSON-safe representation."""
    if isinstance(value, self._json_scalar_types):
        return value

    if isinstance(value, list):
        return [self.load_value(item) for item in value]

    if not isinstance(value, dict):
        raise TypeError(f"Unsupported manifest payload: {type(value).__name__}")

    if value.get(self.marker_key) is True:
        type_name = value.get(_TYPE)
        if type_name == "path":
            return Path(self._require(value, _VALUE, str))
        if type_name == "datetime":
            return datetime.fromisoformat(self._require(value, _VALUE, str))
        if type_name == "uuid":
            return UUID(self._require(value, _VALUE, str))
        if type_name == "tuple":
            items = self._require(value, _VALUE, list)
            return tuple(self.load_value(item) for item in items)
        if type_name == "enum":
            raw = self.load_value(value.get(_VALUE))
            cls = self._import_type(self._require(value, _CLASS, str))
            if cls is None or not issubclass(cls, Enum):
                return raw
            return cls(raw)
        if type_name == "dataclass":
            cls_name = self._require(value, _CLASS, str)
            payload = self._require(value, _FIELDS, dict)
            restored = {key: self.load_value(item) for key, item in payload.items()}
            cls = self._import_type(cls_name)
            if cls is None or not is_dataclass(cls):
                return restored
            return cls(**restored)
        raise TypeError(f"Unknown manifest payload type {type_name!r}")

    return {key: self.load_value(item) for key, item in value.items()}

    def dumps(self, value: Any, *, pretty: bool = False) -> str:
    """Serialise a Python value to JSON using manifest typing."""
    payload = self.dump_value(value)
    if pretty:
        return json.dumps(payload, sort_keys=True, indent=2)
    return json.dumps(payload, sort_keys=True, separators=(",", ":"))
